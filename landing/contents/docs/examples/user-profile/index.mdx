## Complete Example

```jsx
import React from "react";
import { proxy, useStore, useInitSync } from "mesa-react";

// Create a user store
const userStore = proxy({
  user: null,
  lastFetchTime: null,
});

// Mock API function - simulates network request
const fetchUserProfile = (userId) => {
  return new Promise((resolve, reject) => {
    setTimeout(() => {
      // Simulate random success/failure
      if (Math.random() > 0.8) {
        reject(new Error("Network error: Failed to load user profile"));
        return;
      }

      resolve({
        id: userId,
        name: "Sarah Johnson",
        email: "sarah.johnson@example.com",
        avatar: "https://images.unsplash.com/photo-1494790108755-2616b612b77c?w=100&h=100&fit=crop&crop=face",
        bio: "Senior Software Engineer passionate about React and modern web technologies.",
        location: "San Francisco, CA",
        joinDate: "2021-03-15",
        stats: {
          followers: 1284,
          following: 432,
          posts: 156,
        },
        preferences: {
          theme: "dark",
          notifications: true,
          publicProfile: true,
        }
      });
    }, 1500); // Simulate network delay
  });
};

function UserProfile({ userId = 1 }) {
  // Initialize user data with useInitSync
  const { loading, error, refetch } = useInitSync(userStore, async (state) => {
    const userProfile = await fetchUserProfile(userId);
    state.user = userProfile;
    state.lastFetchTime = Date.now();
  }, {
    deps: [userId], // Re-fetch when userId changes
    onSuccess: () => {
      console.log("User profile loaded successfully");
    },
    onError: (err) => {
      console.error("Failed to load user profile:", err);
    }
  });

  // Subscribe to user data only
  const user = useStore(userStore, s => s.user);

  // Loading state
  if (loading) {
    return <LoadingSkeleton />;
  }

  // Error state
  if (error) {
    return <ErrorDisplay error={error} onRetry={refetch} />;
  }

  // No user data
  if (!user) {
    return null;
  }

  return <UserProfileContent user={user} />;
}

// Loading skeleton component
function LoadingSkeleton() {
  return (
    <div className="max-w-2xl mx-auto p-6">
      <div className="animate-pulse">
        <div className="flex items-center space-x-4 mb-6">
          <div className="w-20 h-20 bg-gray-300 rounded-full"></div>
          <div className="space-y-2">
            <div className="h-4 bg-gray-300 rounded w-32"></div>
            <div className="h-3 bg-gray-300 rounded w-48"></div>
          </div>
        </div>
        <div className="space-y-4">
          <div className="h-4 bg-gray-300 rounded w-full"></div>
          <div className="h-4 bg-gray-300 rounded w-3/4"></div>
          <div className="h-4 bg-gray-300 rounded w-1/2"></div>
        </div>
      </div>
    </div>
  );
}

// Error display component
function ErrorDisplay({ error, onRetry }) {
  return (
    <div className="max-w-2xl mx-auto p-6">
      <div className="bg-red-50 border border-red-200 rounded-lg p-4">
        <div className="flex items-center">
          <div className="flex-shrink-0">
            <svg className="h-5 w-5 text-red-400" viewBox="0 0 20 20" fill="currentColor">
              <path fillRule="evenodd" d="M10 18a8 8 0 100-16 8 8 0 000 16zM8.707 7.293a1 1 0 00-1.414 1.414L8.586 10l-1.293 1.293a1 1 0 101.414 1.414L10 11.414l1.293 1.293a1 1 0 001.414-1.414L11.414 10l1.293-1.293a1 1 0 00-1.414-1.414L10 8.586 8.707 7.293z" clipRule="evenodd" />
            </svg>
          </div>
          <div className="ml-3">
            <h3 className="text-sm font-medium text-red-800">
              Error loading profile
            </h3>
            <div className="mt-2 text-sm text-red-700">
              <p>{error}</p>
            </div>
            <div className="mt-4">
              <button
                onClick={onRetry}
                className="bg-red-600 text-white px-4 py-2 rounded hover:bg-red-700 transition-colors"
              >
                Try Again
              </button>
            </div>
          </div>
        </div>
      </div>
    </div>
  );
}

// Main profile content component
function UserProfileContent({ user }) {

  return (
    <div className="max-w-2xl mx-auto p-6">
      <ProfileHeader user={user} />
      <ProfileStats stats={user.stats} />
      <ProfileBio bio={user.bio} location={user.location} joinDate={user.joinDate} />
      <ProfileActions userId={user.id} />
    </div>
  );
}

// Profile header component
function ProfileHeader({ user }) {
  return (
    <div className="flex items-center space-x-4 mb-6">
      <img
        src={user.avatar}
        alt={user.name}
        className="w-20 h-20 rounded-full object-cover border-2 border-gray-300"
      />
      <div>
        <h1 className="text-2xl font-bold text-gray-900">{user.name}</h1>
        <p className="text-gray-600">{user.email}</p>
        <div className="flex items-center mt-2 space-x-4 text-sm text-gray-500">
          <span>📍 {user.location}</span>
          <span>📅 Joined {new Date(user.joinDate).toLocaleDateString()}</span>
        </div>
      </div>
    </div>
  );
}

// Profile statistics
function ProfileStats({ stats }) {
  return (
    <div className="grid grid-cols-3 gap-4 mb-6 text-center">
      <div className="bg-gray-50 p-4 rounded-lg">
        <div className="text-2xl font-bold text-gray-900">{stats.followers}</div>
        <div className="text-sm text-gray-600">Followers</div>
      </div>
      <div className="bg-gray-50 p-4 rounded-lg">
        <div className="text-2xl font-bold text-gray-900">{stats.following}</div>
        <div className="text-sm text-gray-600">Following</div>
      </div>
      <div className="bg-gray-50 p-4 rounded-lg">
        <div className="text-2xl font-bold text-gray-900">{stats.posts}</div>
        <div className="text-sm text-gray-600">Posts</div>
      </div>
    </div>
  );
}

// Profile bio section
function ProfileBio({ bio, location, joinDate }) {
  return (
    <div className="mb-6">
      <h2 className="text-lg font-semibold text-gray-900 mb-2">About</h2>
      <p className="text-gray-700 leading-relaxed">{bio}</p>
    </div>
  );
}

// Profile actions
function ProfileActions({ userId }) {
  const handleFollow = () => {
    alert("Follow functionality would be implemented here");
  };

  const handleMessage = () => {
    alert("Message functionality would be implemented here");
  };

  return (
    <div className="flex space-x-3">
      <button
        onClick={handleFollow}
        className="flex-1 bg-blue-600 text-white py-2 px-4 rounded-lg hover:bg-blue-700 transition-colors"
      >
        Follow
      </button>
      <button
        onClick={handleMessage}
        className="flex-1 border border-gray-300 text-gray-700 py-2 px-4 rounded-lg hover:bg-gray-50 transition-colors"
      >
        Message
      </button>
    </div>
  );
}

export default UserProfile;
```

## Key Implementation Details

### Store Structure

```jsx
const userStore = proxy({
  user: null,              // User data object
  lastFetchTime: null,     // Timestamp of last successful fetch
});
```

### Async Initialization Pattern

```jsx
const { loading, error, refetch } = useInitSync(userStore, async (state) => {
  const userProfile = await fetchUserProfile(userId);
  state.user = userProfile;
  state.lastFetchTime = Date.now();
}, {
  deps: [userId],  // Re-run when dependencies change
  onSuccess: () => console.log("User loaded successfully"),
  onError: (err) => console.error("Failed to load user:", err)
});
```

### Fine-Grained Subscriptions

```jsx
// useInitSync automatically provides loading and error states
const { loading, error, refetch } = useInitSync(/* ... */);

// Only re-renders when user data changes
const user = useStore(userStore, s => s.user);

// Components receive data via props to minimize subscriptions
function UserProfileContent({ user }) {
  return <div>/* ... */</div>;
}
```

## Variations

### With Caching

```jsx
const userCache = new Map();

const fetchUserWithCache = async (userId) => {
  if (userCache.has(userId)) {
    return userCache.get(userId);
  }

  const user = await fetchUserProfile(userId);
  userCache.set(userId, user);
  return user;
};

useInitSync(userStore, async (state) => {
  const user = await fetchUserWithCache(userId);
  state.user = user;
  state.loading = false;
});
```

### With Optimistic Updates

```jsx
const updateUserProfile = async (updates) => {
  // Optimistic update
  const previousUser = userStore.user;
  Object.assign(userStore.user, updates);

  try {
    const updatedUser = await saveUserProfile(userStore.user);
    userStore.user = updatedUser;
  } catch (error) {
    // Rollback on error
    userStore.user = previousUser;
    userStore.error = error.message;
  }
};
```

### With Background Refresh

```jsx
useEffect(() => {
  const interval = setInterval(() => {
    // Refresh user data every 5 minutes without showing loading state
    fetchUserProfile(userId).then(user => {
      userStore.user = user;
    });
  }, 5 * 60 * 1000);

  return () => clearInterval(interval);
}, [userId]);
```

## Best Practices Demonstrated

1. **Automatic State Management**: Let `useInitSync` handle loading and error states automatically
2. **Clean Store Design**: Store only business data, not UI state
3. **Error Recovery**: Provide retry functionality for failed requests
4. **Dependency Management**: Use `deps` array for conditional re-initialization
5. **Fine-Grained Updates**: Subscribe to specific parts of state to minimize re-renders
6. **Prop-Based Data Flow**: Pass data via props to reduce subscription points
7. **User Experience**: Show skeleton loading states instead of blank screens
